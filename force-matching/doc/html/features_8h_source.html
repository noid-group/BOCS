26/* These are defined by the user (or the compiler)
27 to specify the desired environment:
28
29 __STRICT_ANSI__ ISO Standard C.
30 _ISOC99_SOURCE Extensions to ISO C89 from ISO C99.
31 _ISOC11_SOURCE Extensions to ISO C99 from ISO C11.
32 _POSIX_SOURCE IEEE Std 1003.1.
33 _POSIX_C_SOURCE If ==1, like _POSIX_SOURCE; if >=2 add IEEE Std 1003.2;
34 if >=199309L, add IEEE Std 1003.1b-1993;
35 if >=199506L, add IEEE Std 1003.1c-1995;
36 if >=200112L, all of IEEE 1003.1-2004
37 if >=200809L, all of IEEE 1003.1-2008
38 _XOPEN_SOURCE Includes POSIX and XPG things. Set to 500 if
39 Single Unix conformance is wanted, to 600 for the
40 sixth revision, to 700 for the seventh revision.
41 _XOPEN_SOURCE_EXTENDED XPG things and X/Open Unix extensions.
42 _LARGEFILE_SOURCE Some more functions for correct standard I/O.
43 _LARGEFILE64_SOURCE Additional functionality from LFS for large files.
44 _FILE_OFFSET_BITS=N Select default filesystem interface.
45 _BSD_SOURCE ISO C, POSIX, and 4.3BSD things.
46 _SVID_SOURCE ISO C, POSIX, and SVID things.
47 _ATFILE_SOURCE Additional *at interfaces.
48 _GNU_SOURCE All of the above, plus GNU extensions.
49 _REENTRANT Select additionally reentrant object.
50 _THREAD_SAFE Same as _REENTRANT, often used by other systems.
51 _FORTIFY_SOURCE If set to numeric value > 0 additional security
52 measures are defined, according to level.
53
54 The `-ansi' switch to the GNU C compiler defines __STRICT_ANSI__.
55 If none of these are defined, the default is to have _SVID_SOURCE,
56 _BSD_SOURCE, and _POSIX_SOURCE set to one and _POSIX_C_SOURCE set to
57 200112L. If more than one of these are defined, they accumulate.
58 For example __STRICT_ANSI__, _POSIX_SOURCE and _POSIX_C_SOURCE
59 together give you ISO C, 1003.1, and 1003.2, but nothing else.
60
61 These are defined by this file and are used by the
62 header files to decide what to declare or define:
63
64 __USE_ISOC11 Define ISO C11 things.
65 __USE_ISOC99 Define ISO C99 things.
66 __USE_ISOC95 Define ISO C90 AMD1 (C95) things.
67 __USE_POSIX Define IEEE Std 1003.1 things.
68 __USE_POSIX2 Define IEEE Std 1003.2 things.
69 __USE_POSIX199309 Define IEEE Std 1003.1, and .1b things.
70 __USE_POSIX199506 Define IEEE Std 1003.1, .1b, .1c and .1i things.
71 __USE_XOPEN Define XPG things.
72 __USE_XOPEN_EXTENDED Define X/Open Unix things.
73 __USE_UNIX98 Define Single Unix V2 things.
74 __USE_XOPEN2K Define XPG6 things.
75 __USE_XOPEN2KXSI Define XPG6 XSI things.
76 __USE_XOPEN2K8 Define XPG7 things.
77 __USE_XOPEN2K8XSI Define XPG7 XSI things.
78 __USE_LARGEFILE Define correct standard I/O things.
79 __USE_LARGEFILE64 Define LFS things with separate names.
80 __USE_FILE_OFFSET64 Define 64bit interface as default.
81 __USE_BSD Define 4.3BSD things.
82 __USE_SVID Define SVID things.
83 __USE_MISC Define things common to BSD and System V Unix.
84 __USE_ATFILE Define *at interfaces and AT_* constants for them.
85 __USE_GNU Define GNU extensions.
86 __USE_REENTRANT Define reentrant/thread-safe *_r functions.
87 __USE_FORTIFY_LEVEL Additional security measures used, according to level.
88 __FAVOR_BSD Favor 4.3BSD things in cases of conflict.
89
90 The macros `__GNU_LIBRARY__', `__GLIBC__', and `__GLIBC_MINOR__' are
91 defined by this file unconditionally. `__GNU_LIBRARY__' is provided
92 only for compatibility. All new code should use the other symbols
93 to test for features.
94
95 All macros listed above as possibly being defined by this file are
96 explicitly undefined if they are not explicitly defined.
97 Feature-test macros that are not defined by the user or compiler
98 but are implied by the other feature-test macros defined (or by the
99 lack of any definitions) are defined by the file. */

140/* Convenience macros to test the versions of glibc and gcc.
141 Use them like this:
142 #if __GNUC_PREREQ (2,8)
143 ... code requiring gcc 2.8 or later ...
144 #endif
145 Note - they won't work for gcc1 or glibc1, since the _MINOR macros
146 were not defined then. */
147#if defined __GNUC__ && defined __GNUC_MINOR__
148# define __GNUC_PREREQ(maj, min) \
149 ((__GNUC__ << 16) + __GNUC_MINOR__ >= ((maj) << 16) + (min))
150#else
151# define __GNUC_PREREQ(maj, min) 0
152#endif
